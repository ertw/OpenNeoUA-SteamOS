# The shipped Data/OpenNeoUA.ini is embedded as the built-in default configuration.
# It is read again every time CMake configures; a missing file embeds nothing.
set(ONUA_DEFAULT_INI_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../OpenNeoUA.ini")
set(ONUA_DEFAULT_INI_TEXT "")
if(EXISTS "${ONUA_DEFAULT_INI_FILE}")
    file(READ "${ONUA_DEFAULT_INI_FILE}" ONUA_DEFAULT_INI_TEXT)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ONUA_DEFAULT_INI_FILE}")
endif()
# Diagnostics require explicit local opt-in, even when this build machine has
# enabled them in its OpenNeoUA.ini. Do not bake a diagnostic session into defaults.
string(REGEX REPLACE "game\\.crash_diagnostics[ \t]*=[^\r\n]*"
       "game.crash_diagnostics = no" ONUA_DEFAULT_INI_TEXT "${ONUA_DEFAULT_INI_TEXT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/openneoua_default_ini.h"
     "// Generated by CMake from Data/OpenNeoUA.ini. Do not edit.\nstatic const char *const OPENNEOUA_DEFAULT_INI = R\"ONUA(${ONUA_DEFAULT_INI_TEXT})ONUA\";\n")

find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
add_subdirectory(third_party/bullet)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    pkg_check_modules(LIBSYSTEMD QUIET IMPORTED_TARGET libsystemd)
endif()

pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)

pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)

pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)

pkg_check_modules(SDL2_NET REQUIRED IMPORTED_TARGET SDL2_net)

pkg_check_modules(VORBISFILE REQUIRED IMPORTED_TARGET vorbisfile)

# Use the current FFmpeg components; avresample and postproc are obsolete.
pkg_check_modules(
    FFMPEG
    REQUIRED
    IMPORTED_TARGET
    libavformat
    libavcodec
    libavutil
    libswscale
    libswresample
)

find_package(OpenGL REQUIRED)
find_package(OpenAL REQUIRED)
find_package(Lua REQUIRED)

if(WIN32)
    enable_language(RC)
    set(
        WIN32_RESOURCES
        "${CMAKE_CURRENT_SOURCE_DIR}/../svg/icon/icon.rc"
    )

endif()

add_executable(OpenNeoUA ${SOURCES} ${WIN32_RESOURCES})
set_target_properties(OpenNeoUA PROPERTIES WIN32_EXECUTABLE "${WIN32}")

target_link_libraries(
    OpenNeoUA
    PRIVATE

    PkgConfig::SDL2
    PkgConfig::SDL2_IMAGE
    PkgConfig::SDL2_TTF
    PkgConfig::SDL2_NET
    ${OPENGL_LIBRARIES}
    PkgConfig::VORBISFILE
    ${OPENAL_LIBRARY}
    ${LUA_LIBRARIES}

    PkgConfig::FFMPEG
    Threads::Threads
    OpenNeoUACollision
)

if(LIBSYSTEMD_FOUND)
    target_compile_definitions(OpenNeoUA PRIVATE OPENNEOUA_HAVE_SYSTEMD=1)
    target_link_libraries(OpenNeoUA PRIVATE PkgConfig::LIBSYSTEMD)
endif()

if(WIN32)
    target_link_libraries(OpenNeoUA PRIVATE dbghelp psapi)
endif()

install(TARGETS OpenNeoUA RUNTIME DESTINATION bin)

# Keep the private overlay libraries resolvable when the installed executable
# is launched directly or from an AppImage.
if(UNIX AND NOT APPLE)
    set_target_properties(
        OpenNeoUA
        PROPERTIES
        INSTALL_RPATH "$ORIGIN/../lib"
        INSTALL_RPATH_USE_LINK_PATH FALSE
    )
endif()

target_include_directories(
    OpenNeoUA
    PRIVATE

    ${PROJECT_SOURCE_DIR}
    ${CMAKE_CURRENT_BINARY_DIR}/generated
    ${SDL2_INCLUDE_DIRS}
    ${SDL2_IMAGE_INCLUDE_DIRS}
    ${SDL2_TTF_INCLUDE_DIRS}
    ${SDL2_NET_INCLUDE_DIRS}
    ${OpenGL_INCLUDE_DIR}
    ${OPENAL_INCLUDE_DIR}
    ${LUA_INCLUDE_DIR}
)

if(CMAKE_BUILD_TYPE MATCHES [dD][eE][bB][uU][gG])
    target_compile_options(
        OpenNeoUA
        PRIVATE
        "-g"
        "-O0"
    )

    if(NOT WIN32)
        target_compile_options(
            OpenNeoUA
            PRIVATE
            "-fsanitize=address"
        )

        target_link_libraries(
            OpenNeoUA
            PRIVATE
            "-fsanitize=address"
        )
    endif()
endif()
